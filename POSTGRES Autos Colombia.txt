CREATE TABLE tipo_vehiculo (
    id_tipo_vehiculo SERIAL PRIMARY KEY,
    nombre           VARCHAR(20) NOT NULL UNIQUE
);

CREATE TABLE cliente (
    id_cliente     SERIAL PRIMARY KEY,
    documento      VARCHAR(20)  NOT NULL UNIQUE,
    nombres        VARCHAR(60)  NOT NULL,
    apellidos      VARCHAR(60)  NOT NULL,
    telefono       VARCHAR(20),
    correo         VARCHAR(120),
    estado         VARCHAR(10)  NOT NULL DEFAULT 'ACTIVO'
                   CHECK (estado IN ('ACTIVO', 'INACTIVO')),
    fecha_registro TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE vehiculo (
    id_vehiculo      SERIAL PRIMARY KEY,
    placa            VARCHAR(6)  NOT NULL UNIQUE
                     CHECK (placa ~ '^[A-Z]{3}[0-9]{3}$' OR placa ~ '^[A-Z]{3}[0-9]{2}[A-Z]$'),
    marca            VARCHAR(40),
    modelo           VARCHAR(40),
    color            VARCHAR(30),
    id_tipo_vehiculo INT NOT NULL REFERENCES tipo_vehiculo (id_tipo_vehiculo),
    id_cliente       INT NOT NULL REFERENCES cliente (id_cliente)
);

CREATE TABLE celda (
    id_celda         SERIAL PRIMARY KEY,
    codigo           VARCHAR(10) NOT NULL UNIQUE,
    estado           VARCHAR(15) NOT NULL DEFAULT 'DISPONIBLE'
                     CHECK (estado IN ('DISPONIBLE', 'OCUPADA', 'INHABILITADA')),
    id_tipo_vehiculo INT NOT NULL REFERENCES tipo_vehiculo (id_tipo_vehiculo)
);

CREATE TABLE operario (
    id_operario     SERIAL PRIMARY KEY,
    documento       VARCHAR(20)  NOT NULL UNIQUE,
    nombre_completo VARCHAR(120) NOT NULL,
    usuario         VARCHAR(40)  NOT NULL UNIQUE,
    contrasena_hash VARCHAR(100) NOT NULL,
    rol             VARCHAR(15)  NOT NULL DEFAULT 'OPERARIO'
                    CHECK (rol IN ('OPERARIO', 'ADMINISTRADOR')),
    activo          BOOLEAN      NOT NULL DEFAULT TRUE
);

CREATE TABLE registro_acceso (
    id_registro         SERIAL PRIMARY KEY,
    fecha_hora_entrada  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_hora_salida   TIMESTAMP,
    id_vehiculo         INT NOT NULL REFERENCES vehiculo (id_vehiculo),
    id_celda            INT NOT NULL REFERENCES celda (id_celda),
    id_operario_entrada INT NOT NULL REFERENCES operario (id_operario),
    id_operario_salida  INT REFERENCES operario (id_operario),
    CHECK (fecha_hora_salida IS NULL OR fecha_hora_salida >= fecha_hora_entrada),
    CHECK ((fecha_hora_salida IS NULL) = (id_operario_salida IS NULL))
);

-- Un vehículo y una celda solo pueden tener UNA estadía abierta (RNF04)
CREATE UNIQUE INDEX ux_registro_vehiculo_abierto
    ON registro_acceso (id_vehiculo) WHERE fecha_hora_salida IS NULL;
CREATE UNIQUE INDEX ux_registro_celda_abierta
    ON registro_acceso (id_celda) WHERE fecha_hora_salida IS NULL;
CREATE INDEX ix_registro_fecha_entrada ON registro_acceso (fecha_hora_entrada);

CREATE TABLE tipo_novedad (
    id_tipo_novedad SERIAL PRIMARY KEY,
    nombre          VARCHAR(40) NOT NULL UNIQUE
);

CREATE TABLE novedad (
    id_novedad      SERIAL PRIMARY KEY,
    descripcion     VARCHAR(500) NOT NULL CHECK (length(trim(descripcion)) > 0),
    fecha_hora      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    id_registro     INT NOT NULL REFERENCES registro_acceso (id_registro),
    id_tipo_novedad INT NOT NULL REFERENCES tipo_novedad (id_tipo_novedad),
    id_operario     INT NOT NULL REFERENCES operario (id_operario)
);
CREATE INDEX ix_novedad_registro ON novedad (id_registro);
